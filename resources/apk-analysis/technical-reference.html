function getTokenType(scope) {
  if (!scope) return 0;
  var s = scope.replace('hljs-', '');
  return TOKEN_TYPES[s] || 0;
}

// Custom emitter that captures token positions and types
class PositionEmitter {
  constructor(opts) {
    this.options = opts;
    this.tokens = [];
    this.currentPosition = 0;
    this.stack = [];
  }

  addText(text) {
    this.currentPosition += text.length;
  }

  startScope(scope) {
    this.stack.push({
      scope: scope,
      start: this.currentPosition
    });
  }

  endScope() {
    if (this.stack.length === 0) return;
    var item = this.stack.pop();
    var tokenType = getTokenType(item.scope);
    if (tokenType !== 0 && this.currentPosition > item.start) {
      this.tokens.push(item.start, this.currentPosition, tokenType);
    }
  }

  __addSublanguage(emitter, name) {
    if (emitter.tokens) {
      var offset = this.currentPosition;
      for (var i = 0; i < emitter.tokens.length; i += 3) {
        this.tokens.push(
          emitter.tokens[i] + offset,
          emitter.tokens[i + 1] + offset,
          emitter.tokens[i + 2]
        );
      }
    }
    this.currentPosition += emitter.currentPosition;
  }

  openNode(scope) { this.startScope(scope); }
  closeNode() { this.endScope(); }
  toHTML() { return ''; }
  finalize() { return true; }
}

// Entry point called from Android native code
function highlightTokens(code, language) {
  var lang = language || 'plaintext';
  if (lang !== 'plaintext' && !hljs.getLanguage(lang)) {
    lang = 'plaintext';
  }
  const result = hljs.highlight(code, {
    language: lang,
    ignoreIllegals: true
  });
  return JSON.stringify(result._emitter.tokens);
}
The PositionEmitter replaces highlight.js's default HTML emitter. Instead of producing <span> tags, it records flat [start, end, type] triplets. The highlightTokens() function is called from Android via the JS bridge, returning a JSON array that native code parses into ForegroundColorSpan instances.
